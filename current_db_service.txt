30:   Future<void> _createDb(Database db, int version) async {
31:     await db.execute('''
32:       CREATE TABLE notebooks(
33:         id INTEGER PRIMARY KEY AUTOINCREMENT,
34:         name TEXT NOT NULL
35:       )
36:     ''');
37:
38:     await db.execute('''
39:       CREATE TABLE notes(
40:         id INTEGER PRIMARY KEY AUTOINCREMENT,
41:         title TEXT NOT NULL,
42:         content TEXT NOT NULL,
43:         createdAt TEXT NOT NULL,
44:         updatedAt TEXT NOT NULL,
45:         notebookId INTEGER,
46:         FOREIGN KEY (notebookId) REFERENCES notebooks (id) ON DELETE SET NULL
47:       )
48:     ''');
49:
50:     await db.execute('''
51:       CREATE TABLE tags(
52:         id INTEGER PRIMARY KEY AUTOINCREMENT,
53:         name TEXT NOT NULL
54:       )
55:     ''');
56:
57:     await db.execute('''
58:       CREATE TABLE note_tags(
59:         noteId INTEGER,
60:         tagId INTEGER,
61:         PRIMARY KEY (noteId, tagId),
62:         FOREIGN KEY (noteId) REFERENCES notes (id) ON DELETE CASCADE,
